# M2 · S4: Perspective Isolation + @hiphone/perspective Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 把 S3 扁平 key 升级成 owner 维度（`{appId}:owner:{ownerId}:{userKey}`），让 `perspectiveAware: true` 的 user app 在不同 owner 视角下自动数据隔离；`perspectiveAware: false` 的 app 在"查别人手机"视角下显示占位。新增 `@hiphone/perspective` SDK 让 app 查询当前视角。

**Architecture:** 改造 `sdk/storage.ts` 的 key 构造：当前 owner 从 `phoneOwnerStore` 拿（'me' = 玩家, `char-xxx` = 角色）。`globalSet/globalGet` 跨 owner 共享不变。启动 `loadInstalledApps` 前插一次 lazy 迁移（`{appId}:{userKey}` → `{appId}:owner:me:{userKey}`；`{appId}:__global__:{userKey}` → `{appId}:global:{userKey}`）。`@hiphone/perspective` 是现有 `usePerspective()` 的薄 SDK 包装。`AppScene.tsx` 对 user app entry 按 `entry.perspectiveAware` 走 placeholder 逻辑——M1 已经就位，S4 不需要改 AppScene，只需要保证 installer 在注册 Registry 时把 manifest.perspectiveAware 传对。

**Tech Stack:** TypeScript, Vitest, existing `phoneOwnerStore` + `usePerspective`, pnpm.

**Parent spec:** [2026-04-17-m2-user-app-install-design.md](../superpowers/specs/2026-04-17-m2-user-app-install-design.md) §S4 + §2.3 的 S3→S4 迁移部分

```typescript
import { getDB, APP_KV_STORE } from '@/platform/storage/idbStorage';
import type { AppKvRecord } from './appStorage';

/**
 * One-shot, idempotent migration from S3 flat keys to S4 owner-aware
 * keys. Run during app startup before `loadInstalledApps()`.
 *
 * S3 format:
 *   "{appId}:{userKey}"                → scope='app',    ownerId=''
 *   "{appId}:__global__:{userKey}"     → scope='global', ownerId=''
 *
 * S4 format:
 *   "{appId}:owner:me:{userKey}"       → scope='owner',  ownerId='me'
 *   "{appId}:global:{userKey}"         → scope='global', ownerId=''
 *
 * For each row whose value.scope is 'app' or whose value.scope is
 * 'global' with an `__global__` infix key, rewrite to the S4 key
 * and update the value's scope/ownerId fields. Records already in
 * S4 shape are left untouched (idempotent).
 */
export async function migrateS3ToS4Owner(): Promise<void> {
  const db = await getDB();
  const migrations: Array<{ oldKey: string; newKey: string; newValue: AppKvRecord }> = [];

  await new Promise<void>((resolve, reject) => {
    const tx = db.transaction(APP_KV_STORE, 'readonly');
    const store = tx.objectStore(APP_KV_STORE);
    const req = store.openCursor();
    req.onsuccess = () => {
      const cursor = req.result;
      if (!cursor) return;
      const key = String(cursor.key);
      const record = cursor.value as AppKvRecord;

      if (record.scope === 'app' && record.ownerId === '') {
        // S3 per-app flat → S4 owner:me
        const newKey = `${record.appId}:owner:me:${record.userKey}`;
        const newValue: AppKvRecord = {
          ...record, scope: 'owner', ownerId: 'me',
        };
        migrations.push({ oldKey: key, newKey, newValue });
      } else if (record.scope === 'global' && key.includes(':__global__:')) {
        // S3 global with __global__ infix → S4 global (no __global__)
        const newKey = `${record.appId}:global:${record.userKey}`;
        const newValue: AppKvRecord = { ...record };
        migrations.push({ oldKey: key, newKey, newValue });
      }
      // else: already S4 shape (scope='owner' with ownerId populated, or
      //       scope='global' with non-__global__ key format) — leave alone.

      cursor.continue();
    };
    tx.oncomplete = () => resolve();
    tx.onerror = () => reject(tx.error);
  });

  if (migrations.length === 0) return;

  await new Promise<void>((resolve, reject) => {
    const tx = db.transaction(APP_KV_STORE, 'readwrite');
    const store = tx.objectStore(APP_KV_STORE);
    for (const m of migrations) {
      store.delete(m.oldKey);
      store.put(m.newValue, m.newKey);
    }
    tx.oncomplete = () => resolve();
    tx.onerror = () => reject(tx.error);
  });
}
```

```typescript
import {
  appStorageGet,
  appStorageSet,
  appStorageRemove,
  appStorageListByAppId,
  type AppKvRecord,
} from '../appStorage';
import { getCurrentAppId } from './context';
import { usePhoneOwnerStore } from '@/platform/stores/phoneOwnerStore';

/**
 * @hiphone/storage — user-app-facing key-value storage.
 *
 * Two namespaces:
 *   - Per-owner:   set/get/remove/list — `{appId}:owner:{ownerId}:{key}`
 *   - Global:      globalSet/globalGet — `{appId}:global:{key}`
 *
 * ownerId:
 *   - 'me' when player is viewing their own phone
 *   - 'char-{id}' when player is viewing a character's phone
 */

function currentOwnerId(): string {
  const id = usePhoneOwnerStore.getState().phoneOwnerId;
  return id === null ? 'me' : `char-${id}`;
}

const OWNER_PREFIX = (appId: string, ownerId: string) =>
  `${appId}:owner:${ownerId}:`;
const GLOBAL_PREFIX = (appId: string) => `${appId}:global:`;

export async function get(key: string): Promise<unknown> {
  const appId = getCurrentAppId();
  const ownerId = currentOwnerId();
  const fullKey = OWNER_PREFIX(appId, ownerId) + key;
  const record = await appStorageGet(appId, fullKey);
  return record?.value;
}

export async function set(key: string, value: unknown): Promise<void> {
  const appId = getCurrentAppId();
  const ownerId = currentOwnerId();
  const fullKey = OWNER_PREFIX(appId, ownerId) + key;
  const record: AppKvRecord = {
    appId, scope: 'owner', ownerId, userKey: key, value,
  };
  await appStorageSet(appId, fullKey, record);
}

export async function remove(key: string): Promise<void> {
  const appId = getCurrentAppId();
  const ownerId = currentOwnerId();
  const fullKey = OWNER_PREFIX(appId, ownerId) + key;
  await appStorageRemove(appId, fullKey);
}

export async function list(): Promise<string[]> {
  const appId = getCurrentAppId();
  const ownerId = currentOwnerId();
  const ownerPrefix = OWNER_PREFIX(appId, ownerId);
  const fullKeys = await appStorageListByAppId(appId);
  const out: string[] = [];
  for (const k of fullKeys) {
    if (k.startsWith(ownerPrefix)) out.push(k.slice(ownerPrefix.length));
  }
  return out;
}

export async function globalGet(key: string): Promise<unknown> {
  const appId = getCurrentAppId();
  const fullKey = GLOBAL_PREFIX(appId) + key;
  const record = await appStorageGet(appId, fullKey);
  return record?.value;
}

export async function globalSet(key: string, value: unknown): Promise<void> {
  const appId = getCurrentAppId();
  const fullKey = GLOBAL_PREFIX(appId) + key;
  const record: AppKvRecord = {
    appId, scope: 'global', ownerId: '', userKey: key, value,
  };
  await appStorageSet(appId, fullKey, record);
}
```

```typescript
import { useMemo } from 'react';
import { usePhoneOwnerStore } from '@/platform/stores/phoneOwnerStore';

export interface CurrentOwner {
  /** 'me' when player; 'char-{id}' when viewing another's phone. */
  ownerId: string;
  /** Human-readable label for UI (matches ownerId when no nicer name available). */
  ownerName: string;
  isViewingOther: boolean;
}

function snapshot(): CurrentOwner {
  const phoneOwnerId = usePhoneOwnerStore.getState().phoneOwnerId;
  if (phoneOwnerId === null) {
    return { ownerId: 'me', ownerName: '我', isViewingOther: false };
  }
  return {
    ownerId: `char-${phoneOwnerId}`,
    ownerName: `char-${phoneOwnerId}`,
    isViewingOther: true,
  };
}

/**
 * React hook — reactive version that re-renders when perspective
 * changes. Safe to call inside user app components.
 */
export function useCurrentOwner(): CurrentOwner {
  const phoneOwnerId = usePhoneOwnerStore((s) => s.phoneOwnerId);
  return useMemo(() => {
    if (phoneOwnerId === null) {
      return { ownerId: 'me', ownerName: '我', isViewingOther: false };
    }
    return {
      ownerId: `char-${phoneOwnerId}`,
      ownerName: `char-${phoneOwnerId}`,
      isViewingOther: true,
    };
  }, [phoneOwnerId]);
}

/**
 * One-shot non-reactive read. Useful in event handlers, async
 * callbacks, or non-React contexts.
 */
export function getCurrentOwner(): CurrentOwner {
  return snapshot();
}
```
