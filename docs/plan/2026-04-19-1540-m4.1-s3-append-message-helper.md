# M4.1 S3 — `_appendMessage` Helper + Write-Path Unification

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Introduce the platform-level `_appendMessage(msg, source)` helper that simultaneously writes to `xingYuDataStore.messages[]` AND `characterMemoryStore.entries[charId]`. Migrate every `setState({ messages: [...] })` call site in the codebase to use it. **After this stage, memoryStore gets populated on every write; but no reader consumes it yet (`collectCharacterHistory` and `promptAssembly` still use the old path).**

**Architecture:** `_appendMessage` lives in `src/platform/ai/memoryWriter.ts`. It (1) computes the new xingYuDataStore state (messages[] append + conversation lastMsg/lastTime/unread update, preserving current semantics), (2) calls `buildMemoryEntry` for a text mirror, (3) appends to memoryStore for the current conv's characterId. For AI-AI conversations (`conv.aiChatParticipants?.length > 0`), it also appends a parallel entry to each other participant's memoryStore with role flipped from their perspective.

**Tech Stack:** TypeScript, Vitest. Depends on S1 `characterMemoryStore` and S2 `buildMemoryEntry`.

```typescript
// src/platform/ai/memoryWriter.ts
import type { Conversation, Message } from '@/apps/XingYu/data';
import { useXYData } from '@/apps/XingYu/xingYuDataStore';
import { useCharacterStore } from '@/platform/stores/characterStore';
import { usePersonaStore } from '@/platform/stores/personaStore';
import { useCharacterMemory } from './characterMemoryStore';
import type { MemorySource } from './characterMemoryStore';
import { buildMemoryEntry, type BuildMemoryContext } from './buildMemoryEntry';
import { useXYNav } from '@/apps/XingYu/xingYuNavStore';

/**
 * Canonical write path: update xingYuDataStore.messages[] + lastMsg/lastTime/unread
 * AND mirror the message into characterMemoryStore (and into AI-AI participants).
 */
export function _appendMessage(msg: Message, source: MemorySource): void {
  // ── 1. xingYuDataStore write ──────────────────────────────
  const xy = useXYData.getState();
  const activeConvId = useXYNav.getState().activeChatId;
  useXYData.setState({
    messages: [...xy.messages, msg],
    conversations: xy.conversations.map((c) =>
      c.id === msg.convId
        ? {
            ...c,
            lastMsg: derivePreview(msg),
            lastTime: msg.timestamp,
            unread: shouldIncrementUnread(msg, c, activeConvId)
              ? c.unread + 1
              : c.unread,
          }
        : c,
    ),
  });

  // ── 2. memoryStore write (for the conv's primary character) ─
  const charId = deriveCharacterIdFromConv(msg.convId);
  if (!charId) return;

  const ctx = buildMemoryContextFor(charId);
  const entry = buildMemoryEntry(msg, source, ctx);
  if (entry) {
    useCharacterMemory.getState().append(charId, entry);
  }

  // ── 3. AI-AI cross-write for every other participant ───────
  const conv = useXYData.getState().conversations.find((c) => c.id === msg.convId);
  if (conv?.aiChatParticipants?.length) {
    for (const otherId of conv.aiChatParticipants) {
      if (otherId === charId) continue;
      const otherCtx = buildMemoryContextFor(otherId);
      const otherEntry = buildMemoryEntry(msg, source, otherCtx);
      if (otherEntry) {
        useCharacterMemory.getState().append(otherId, otherEntry);
      }
    }
  }
}

// ════════════════════════════════════════════════════════════════
// Helpers
// ════════════════════════════════════════════════════════════════

function derivePreview(msg: Message): string {
  switch (msg.type) {
    case 'text':
      return (msg.text ?? '').slice(0, 60);
    case 'image':
      return '[图片]';
    case 'sticker':
      return msg.stickerDesc ? `[表情：${msg.stickerDesc}]` : '[表情]';
    case 'forward_card':
      return `[聊天记录：${msg.forwardCard.title}]`;
    case 'heartbeat_log':
      return (msg.text ?? '').slice(0, 60);
    default:
      return '';
  }
}

function shouldIncrementUnread(
  msg: Message,
  conv: Conversation,
  activeConvId: string | null,
): boolean {
  // Never bump for the user's own messages, for already-active chat, or for heartbeat_log.
  if (msg.senderId === 'me') return false;
  if (activeConvId === conv.id) return false;
  if (msg.type === 'heartbeat_log') return false;
  return true;
}

function deriveCharacterIdFromConv(convId: string): string | null {
  // Primary user↔character conv: "c-char-<id>"
  if (convId.startsWith('c-char-')) {
    return convId.slice('c-char-'.length);
  }
  // AI-AI conv: convId is arbitrary; the aiChatParticipants array drives it. In
  // that case, we arbitrarily "anchor" on the first participant; the cross-write
  // logic will fan out to the rest.
  const conv = useXYData.getState().conversations.find((c) => c.id === convId);
  if (conv?.aiChatParticipants?.length) return conv.aiChatParticipants[0]!;
  return null;
}

function buildMemoryContextFor(currentCharId: string): BuildMemoryContext {
  const characters = useCharacterStore.getState().characters;
  const persona = usePersonaStore.getState().getActivePersona();
  const userNickname = useXYData.getState().userSettings?.nickname ?? '用户';
  return {
    currentCharId,
    charactersById: new Map(characters.map((c) => [c.id, { id: c.id, name: c.name }])),
    personaName: persona?.name ?? userNickname,
    userNickname,
  };
}
```

```typescript
import { _appendMessage } from '@/platform/ai/memoryWriter';

function execSendMessage(input, characterId, _store): ToolResult {
  const text = String(input.text ?? '').trim();
  if (!text) return { observation: '消息内容不能为空。', done: false };

  const convId = `c-char-${characterId}`;
  const senderId = `char-${characterId}`;
  const now = Date.now();

  const dataState = useXYData.getState();
  let conv = dataState.conversations.find((c) => c.id === convId);

  // Ensure conversation exists (first-time heartbeat-initiated send).
  if (!conv) {
    const character = useCharacterStore.getState().characters.find((c) => c.id === characterId);
    if (!character) return { observation: '找不到对应角色。', done: false };
    conv = {
      id: convId, idolId: senderId, characterId,
      lastMsg: '', lastTime: now, unread: 0,
    };
    useXYData.setState({ conversations: [conv, ...dataState.conversations] });
  }

  const msg: Message = {
    id: uid(), convId, senderId, type: 'text', text,
    timestamp: now, proactive: true,
  };

  _appendMessage(msg, 'heartbeat');

  useHeartbeatStore.getState().pushLog({
    characterId, action: 'send_message', detail: text.slice(0, 40),
  });
  return { observation: '消息已发送。', done: false };
}
```
