# M4.1 S6 — Summarizer Migration (conv-level → character-level)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Retarget compression from per-conversation (stored on `Conversation.summary` / `summaryUpToTimestamp`) to per-character (stored inline in `characterMemoryStore.entries[charId]` as a `role:'system', compressed:true` MemoryEntry). Triggering moves from "XingYu calls triggerCompression after each reply" to "memoryStore.append internally checks token budget and schedules compression". XingYu / heartbeat / aiChat lose their direct interest — this becomes an invisible platform service.

**Architecture:**
1. New async platform function `runCompressionIfNeeded(characterId)` in `src/platform/ai/characterMemoryCompression.ts`. Reads the character's current entries + aiConfig thresholds, computes token count, selects the oldest-but-not-recent batch, calls the existing `compressHistory` (reused from `summarizer.ts`) to produce a summary string, and calls `memoryStore.replaceRange()` with a new `role:'system', compressed:true` entry.
2. `characterMemoryStore.append` subscribes to a post-append hook: after every append, schedule `runCompressionIfNeeded` on a microtask (debounce to avoid firing on every append during batchAppend). Module-level in-flight Map per charId to prevent concurrent compression runs.
3. Delete `triggerCompression` export from `xingYuDataStore.ts`. Delete `Conversation.summary` / `summaryUpToTimestamp` fields (legacy data left behind safely — promptAssembly no longer reads them).
4. Existing `compressHistory` in `summarizer.ts` — keep, reuse. Its input is `messagesToCompress: ChatMessage[]`; adapt by first rendering our `MemoryEntry[]` slice through `renderMemoryToChatMessages` (S4's function) as the compressHistory input.

**Tech Stack:** TypeScript, Vitest, existing `compressHistory`. No new deps.

// src/platform/ai/characterMemoryCompression.ts
import { useAIConfigStore } from '@/platform/stores/aiConfigStore';
import { useCharacterStore } from '@/platform/stores/characterStore';
import { usePersonaStore } from '@/platform/stores/personaStore';
import { getAdapter } from './providers';
import { compressHistory } from './summarizer';
import {
  useCharacterMemory,
  type MemoryEntry,
} from './characterMemoryStore';
import { estimateTokens } from './tokenEstimator';
import { uid } from '@/platform/utils/uid';

const inFlight = new Map<string, Promise<void>>();

export function runCompressionIfNeeded(characterId: string): Promise<void> {
  const existing = inFlight.get(characterId);
  if (existing) return existing;

  const p = (async () => {
    try {
      await doCompression(characterId);
    } catch (e) {
      console.warn(`[compression] ${characterId} failed:`, e);
    }
  })();
  inFlight.set(characterId, p);
  try {
    await_no_op; // ensure the in-flight map is released even if nothing awaits us
  } catch { /* not reachable */ }
  p.finally(() => {
    if (inFlight.get(characterId) === p) inFlight.delete(characterId);
  });
  return p;
}

// Sentinel so TS doesn't complain; refactored out in Step 4.
declare const await_no_op: void;

async function doCompression(characterId: string): Promise<void> {
  const entries = useCharacterMemory.getState().getAll(characterId);
  if (entries.length === 0) return;

  const aiConfig = useAIConfigStore.getState();
  if (!aiConfig.apiKey) return;

  const adapter = getAdapter(aiConfig.provider);
  if (!adapter) return;

  const threshold = aiConfig.summarizeThreshold ?? 0;
  if (threshold <= 0) return;

  // Count tokens excluding already-compressed entries
  const liveEntries = entries.filter((e) => !e.compressed);
  const tokens = liveEntries.reduce(
    (sum, e) => sum + estimateTokens(e.content) + 6,
    0,
  );
  const budget = aiConfig.contextWindow * threshold;
  if (tokens <= budget) return;

  // Find the slice to compress: oldest → up to entries.length - keepRecent
  const keepRecent = aiConfig.keepRecentMessages;
  const compressEndIdx = entries.length - keepRecent - 1;
  if (compressEndIdx < 0) return;

  // Don't re-compress already-compressed entries
  const compressStartIdx = entries.findIndex((e) => !e.compressed);
  if (compressStartIdx < 0 || compressStartIdx > compressEndIdx) return;

  const slice = entries.slice(compressStartIdx, compressEndIdx + 1);
  if (slice.length === 0) return;

  const startEntry = slice[0]!;
  const endEntry = slice[slice.length - 1]!;

  // Map each entry into compressHistory's expected shape
  const messagesToCompress = slice.map((e) => ({
    role: e.role === 'assistant' ? ('assistant' as const) : ('user' as const),
    content: e.content,
  }));

  const character = useCharacterStore.getState().characters.find((c) => c.id === characterId);
  const persona = usePersonaStore.getState().getActivePersona();

  // Preserve any previous summary (the most recent compressed entry, if any)
  const previousSummary = [...entries]
    .reverse()
    .find((e) => e.compressed)?.content;

  const summaryText = await compressHistory({
    previousSummary,
    messagesToCompress,
    endpoint: aiConfig.apiEndpoint || adapter.defaultEndpoint,
    apiKey: aiConfig.apiKey,
    model: aiConfig.model,
    providerId: aiConfig.provider,
    characterName: character?.name ?? '角色',
    userName: persona?.name ?? '用户',
    contextWindow: aiConfig.contextWindow,
    maxTokens: aiConfig.maxTokens,
  });

  const summaryEntry: MemoryEntry = {
    id: uid(),
    characterId,
    role: 'system',
    speakerId: 'system',
    content: `[之前的对话摘要]\n${summaryText}`,
    source: 'system',
    createdAt: endEntry.createdAt + 1,
    compressed: true,
  };

  useCharacterMemory
    .getState()
    .replaceRange(characterId, startEntry.id, endEntry.id, summaryEntry);
}
```

type PostAppendHook = (characterId: string) => void;
let postAppendHook: PostAppendHook | null = null;
export function setPostAppendHook(hook: PostAppendHook | null): void {
  postAppendHook = hook;
}

// Invoke inside append / batchAppend after state is updated:
append(characterId, input) {
  const entry = makeEntry(characterId, input);
  set((state) => ({
    entries: {
      ...state.entries,
      [characterId]: [...(state.entries[characterId] ?? []), entry],
    },
  }));
  if (postAppendHook) {
    queueMicrotask(() => postAppendHook!(characterId));
  }
  return entry;
},

// Similarly in batchAppend — fire the hook once after the whole batch.
batchAppend(characterId, inputs) {
  const newEntries = inputs.map((i) => makeEntry(characterId, i));
  set((state) => ({
    entries: {
      ...state.entries,
      [characterId]: [...(state.entries[characterId] ?? []), ...newEntries],
    },
  }));
  if (postAppendHook) {
    queueMicrotask(() => postAppendHook!(characterId));
  }
  return newEntries;
},
```
