# M4.1 S8 — `@hiphone/ai` Level 2 (ChatSession)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add `chatWithCharacter(charId, opts): ChatSession` to `@hiphone/ai`. Session supports `send` / `streamSend` / `append` / `replyToLast` / `streamReplyToLast` / `abort` / `history`, with `persistent: true|false` and `mirror: true|false` semantics per the spec §4 rules table. Fixture `ai-chat-app` exercises the API.

**Architecture:** Session is a plain object returned by `chatWithCharacter`. Internally: holds a session-local buffer (`_buffer: SessionEntry[]`), an AbortController per in-flight call, and a reference to the originating appId (captured at creation via `getCurrentAppId()`). On every `send` / `replyToLast` path:
1. Build the full chat history = memoryStore entries + session buffer (for persistent=false: snapshot taken at creation replaces the memoryStore read).
2. Call `assemblePrompt` to get ChatMessage[] for the LLM; also inject the just-`append`ed user message as the final role:user trigger.
3. Call `chatComplete`, get raw reply.
4. Update session buffer + memoryStore per §4 rules.

**Tech Stack:** TypeScript, Vitest. Depends on S1-S7.

```typescript
import { getCurrentAppId } from './context';
import { useCharacterMemory } from '@/platform/ai/characterMemoryStore';
import type { MemoryEntry } from '@/platform/ai/characterMemoryStore';

// ════════════════════════════════════════════════════════════════
// Level 2 — character chat session
// ════════════════════════════════════════════════════════════════

export interface ChatOptions {
  persistent?: boolean;
  signal?: AbortSignal;
}

export interface SessionEntry {
  role: 'user' | 'assistant';
  speakerId: string;
  content: string;
  timestamp: number;
}

export interface MirrorOption {
  mirror?: boolean;
}

export interface ChatSession {
  readonly characterId: string;
  readonly persistent: boolean;
  readonly history: readonly SessionEntry[];
  send(content: string, opts?: MirrorOption): Promise<string>;
  streamSend(content: string, opts?: MirrorOption): AsyncIterable<string>;
  append(
    entry: { role: 'user' | 'assistant'; content: string; speakerId?: string },
    opts?: MirrorOption,
  ): void;
  replyToLast(opts?: MirrorOption): Promise<string>;
  streamReplyToLast(opts?: MirrorOption): AsyncIterable<string>;
  abort(): void;
}

export function chatWithCharacter(
  characterId: string,
  options: ChatOptions = {},
): ChatSession {
  const character = useCharacterStore
    .getState()
    .characters.find((c) => c.id === characterId);
  if (!character) throw new AICharacterNotFoundError(characterId);

  const persistent = options.persistent === true;
  const callerAppId = getCurrentAppId(); // null-safe; user apps always have an appId
  const sessionAppId = callerAppId ?? 'unknown';

  // Snapshot for persistent=false: freeze what the character "remembers" right now.
  const snapshot = !persistent
    ? useCharacterMemory.getState().getSnapshot(characterId)
    : null;

  const buffer: SessionEntry[] = [];
  const controllers: AbortController[] = [];

  function append(
    entry: { role: 'user' | 'assistant'; content: string; speakerId?: string },
    opts: MirrorOption = {},
  ): void {
    const mirror = opts.mirror !== false; // default true
    const speakerId =
      entry.speakerId ??
      (entry.role === 'user' ? 'me' : characterId);

    buffer.push({
      role: entry.role,
      speakerId,
      content: entry.content,
      timestamp: Date.now(),
    });

    if (persistent && mirror) {
      useCharacterMemory.getState().append(characterId, {
        role: entry.role,
        speakerId,
        content: entry.content,
        source: `app:${sessionAppId}` as const,
      });
    }
  }

  function abort(): void {
    for (const c of controllers) c.abort();
    controllers.length = 0;
  }

  // send / streamSend / replyToLast implemented in next task.
  function send(_content: string, _opts?: MirrorOption): Promise<string> {
    throw new Error('TODO: Task 2');
  }
  function streamSend(_content: string, _opts?: MirrorOption): AsyncIterable<string> {
    throw new Error('TODO: Task 2');
  }
  function replyToLast(_opts?: MirrorOption): Promise<string> {
    throw new Error('TODO: Task 2');
  }
  function streamReplyToLast(_opts?: MirrorOption): AsyncIterable<string> {
    throw new Error('TODO: Task 2');
  }

  return {
    characterId,
    persistent,
    get history() {
      return buffer as readonly SessionEntry[];
    },
    send,
    streamSend,
    append,
    replyToLast,
    streamReplyToLast,
    abort,
  };
}
```

```typescript
import { useCharacterStore } from '@/platform/stores/characterStore';
import { usePersonaStore } from '@/platform/stores/personaStore';
import { useWorldBookStore } from '@/platform/stores/worldBookStore';
import { assemblePrompt } from '@/platform/ai/promptAssembly';
import { buildDeviceContext } from '@/platform/ai/deviceContext';
import { useStickerStore } from '@/apps/XingYu/stickerStore';

async function callLLM(
  currentCharId: string,
  sessionBuffer: readonly SessionEntry[],
  persistent: boolean,
  snapshot: readonly MemoryEntry[] | null,
  signal: AbortSignal,
): Promise<string> {
  const provider = requireProvider();
  const character = useCharacterStore
    .getState()
    .characters.find((c) => c.id === currentCharId);
  if (!character) throw new AICharacterNotFoundError(currentCharId);

  const persona = usePersonaStore.getState().getActivePersona();
  const worldBookChunk = useWorldBookStore.getState().buildSystemPromptChunk();
  const aiConfig = useAIConfigStore.getState();
  const allCharacters = useCharacterStore.getState().characters;
  const charactersById = new Map(allCharacters.map((c) => [c.id, { id: c.id, name: c.name }]));
  const allStickers = useStickerStore.getState().packs.flatMap((pack) =>
    pack.stickers.map((s) => ({ id: s.id, description: s.description })),
  );

  // Merge source: persistent=true reads live memoryStore; persistent=false uses snapshot.
  // Buffer entries that are NOT already in memoryStore (mirror=false) need to be appended on top.
  const liveBase = persistent
    ? useCharacterMemory.getState().getAll(currentCharId)
    : snapshot!;

  const bufferAsEntries: MemoryEntry[] = sessionBuffer.map((b, i) => ({
    id: `buf-${i}`,
    characterId: currentCharId,
    role: b.role,
    speakerId: b.speakerId,
    content: b.content,
    source: 'system',
    createdAt: b.timestamp,
  }));

  // For persistent=true + mirror=true session flows, buffer entries may duplicate
  // liveBase entries. De-dup by (role, speakerId, content, timestamp).
  const memoryEntries = [...liveBase];
  for (const be of bufferAsEntries) {
    const already = memoryEntries.some(
      (m) => m.createdAt === be.createdAt && m.content === be.content && m.speakerId === be.speakerId,
    );
    if (!already) memoryEntries.push(be);
  }

  const { messages: chatMessages } = assemblePrompt({
    character: {
      name: character.name,
      description: character.description,
      personality: character.personality,
      scenario: character.scenario,
      systemPrompt: character.systemPrompt,
      postHistoryInstructions: character.postHistoryInstructions,
      messageExamples: character.messageExamples,
    },
    persona: { name: persona?.name ?? '用户', description: persona?.description ?? '' },
    aiConfig: {
      systemPrompt: aiConfig.systemPrompt,
      postHistoryInstructions: aiConfig.postHistoryInstructions,
      contextWindow: aiConfig.contextWindow,
      maxTokens: aiConfig.maxTokens,
      keepRecentMessages: aiConfig.keepRecentMessages,
      worldInfoBudgetPercent: aiConfig.worldInfoBudgetPercent,
      enableVision: aiConfig.enableVision,
    },
    worldBookChunk,
    memoryEntries,
    currentCharId,
    charactersById,
    now: new Date(),
    deviceContext: buildDeviceContext(),
    availableStickers: allStickers.length > 0 ? allStickers : undefined,
  });

  return chatComplete(
    { endpoint: provider.endpoint, apiKey: provider.apiKey, model: provider.model, providerId: provider.providerId },
    chatMessages,
    { maxTokens: provider.maxTokens, temperature: provider.temperature },
    signal,
  );
}

async function send(content: string, opts: MirrorOption = {}): Promise<string> {
  const mirror = opts.mirror !== false;
  const controller = new AbortController();
  controllers.push(controller);
  try {
    // Step 1: enqueue the user message.
    append({ role: 'user', content }, { mirror });

    // Step 2: call LLM.
    const reply = await callLLM(characterId, buffer, persistent, snapshot, controller.signal);

    // Step 3: enqueue the assistant reply.
    append({ role: 'assistant', content: reply, speakerId: characterId }, { mirror });
    return reply;
  } finally {
    const idx = controllers.indexOf(controller);
    if (idx >= 0) controllers.splice(idx, 1);
  }
}

async function replyToLast(opts: MirrorOption = {}): Promise<string> {
  const mirror = opts.mirror !== false;
  const controller = new AbortController();
  controllers.push(controller);
  try {
    const reply = await callLLM(characterId, buffer, persistent, snapshot, controller.signal);
    append({ role: 'assistant', content: reply, speakerId: characterId }, { mirror });
    return reply;
  } finally {
    const idx = controllers.indexOf(controller);
    if (idx >= 0) controllers.splice(idx, 1);
  }
}
```

```typescript
async function* streamSend(content: string, opts: MirrorOption = {}): AsyncIterable<string> {
  const mirror = opts.mirror !== false;
  const controller = new AbortController();
  controllers.push(controller);
  try {
    append({ role: 'user', content }, { mirror });
    const reply = await callLLM(characterId, buffer, persistent, snapshot, controller.signal);
```

```tsx
import React, { useState, useMemo } from 'react';
import { MessageCircle } from 'lucide-react';
import {
  chatWithCharacter,
  getCharacters,
  extractPlainText,
  type ChatSession,
  type CharacterInfo,
} from '@hiphone/ai';

type Msg = { role: 'user' | 'assistant'; text: string };

export default function AIChatApp() {
  const characters: CharacterInfo[] = useMemo(() => getCharacters(), []);
  const [charId, setCharId] = useState(characters[0]?.id ?? '');
  const [persistent, setPersistent] = useState(false);
  const [session, setSession] = useState<ChatSession | null>(null);
  const [msgs, setMsgs] = useState<Msg[]>([]);
  const [input, setInput] = useState('');
  const [busy, setBusy] = useState(false);

  const start = () => {
    if (!charId) return;
    const s = chatWithCharacter(charId, { persistent });
    setSession(s);
    setMsgs([]);
  };

  const sendOne = async () => {
    if (!session || !input.trim()) return;
    const text = input;
    setInput('');
    setMsgs((m) => [...m, { role: 'user', text }]);
    setBusy(true);
    try {
      const raw = await session.send(text);
      const plain = extractPlainText(raw);
      setMsgs((m) => [...m, { role: 'assistant', text: plain }]);
    } catch (e) {
      setMsgs((m) => [...m, { role: 'assistant', text: `⚠️ ${e instanceof Error ? e.message : String(e)}` }]);
    } finally {
      setBusy(false);
    }
  };
```
