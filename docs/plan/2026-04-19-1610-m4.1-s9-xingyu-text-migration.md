# M4.1 S9 — XingYu Text-Chat Migration to Session

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Migrate XingYu's text-message AI reply flow from direct `assemblePrompt` + `chatComplete` to `@hiphone/ai.chatWithCharacter` session. parseReply / filterReply / multi-bubble delivery / 300-800ms intervals / mock fallback (`IDOL_REPLY_POOL`) all preserved. AbortController becomes session.abort. memoryStore gets a single raw-JSON assistant entry per reply.

**Architecture:** `scheduleAICharacterReply` in `xingYuDataStore.ts` is rewritten. Per reply:
1. Look up conv + character.
2. `const session = chatWithCharacter(charId, { persistent: true })`.
3. `session.append({ role: 'user', content: text }, { mirror: false })` — the user msg was already written via `_appendMessage` in S3.
4. `for await (const chunk of session.streamReplyToLast({ mirror: false }))` accumulate raw reply.
5. `parseReply(rawReply).map(filterReply)` → bubbles.
6. Deliver bubbles via `_appendMessage(bubbleMsg, 'xingyu')` with inter-bubble delays.
7. Write ONE raw assistant entry to memoryStore (`useCharacterMemory.append(...)` directly, bypassing `_appendMessage` to avoid per-bubble pollution).

AbortController replaced: `aiControllers: Map<convId, AbortController>` → `aiSessions: Map<convId, ChatSession>`.

**Tech Stack:** TypeScript, Vitest.

// src/apps/XingYu/xingYuDataStore.ts

import {
  chatWithCharacter,
  AIUnavailableError,
  type ChatSession,
} from '@/platform/userApp/sdk/ai';
import { useCharacterMemory } from '@/platform/ai/characterMemoryStore';

// Replace `aiControllers: Map<convId, AbortController>` with
// `aiSessions: Map<convId, ChatSession>`
const aiSessions = new Map<string, ChatSession>();

function scheduleAICharacterReply(convId: string, get: () => XingYuDataState) {
  const state = get();
  const conv = state.conversations.find((c) => c.id === convId);
  if (!conv?.characterId) return;
  const characterId = conv.characterId;

  // Legacy mock-idol convs → no character, fallback pool path (unchanged).
  const character = useCharacterStore.getState().characters.find((c) => c.id === characterId);
  if (!character) return;

  // Insert typing indicator placeholder (transient — NOT via _appendMessage; it
  // gets filtered out when the reply arrives).
  const placeholderId = uid();
  const placeholderMsg: Message = {
    id: placeholderId, convId,
    senderId: `char-${characterId}`,
    type: 'text', text: '', timestamp: Date.now(),
    streaming: true,
  };
  useXYData.setState({
    messages: [...useXYData.getState().messages, placeholderMsg],
  });

  // Create session; capture the most-recent user message's text so we can
  // session.append() it (without re-writing to memoryStore — S3 already wrote).
  const session = chatWithCharacter(characterId, { persistent: true });
  aiSessions.set(convId, session);

  // Find the latest user text relative to convId (there may be multiple
  // pending messages — send all of them to the session buffer).
  const recentUserMsgs = useXYData
    .getState()
    .messages
    .filter((m) =>
      m.convId === convId && m.senderId === 'me' &&
      !m.streaming && m.timestamp < placeholderMsg.timestamp,
    )
    .slice(-1); // last one; earlier ones are already in memoryStore from S3

  for (const um of recentUserMsgs) {
    const content = um.type === 'text' ? (um.text ?? '') :
                    um.type === 'image' ? `[图片 ${um.imageUrl}]` :
                    um.type === 'sticker' ? `[表情：${um.stickerDesc ?? ''}]` :
                    um.type === 'forward_card' ? `[转发...]` :
                    '';
    session.append({ role: 'user', content }, { mirror: false });
  }

  void (async () => {
    let rawReply = '';
    try {
      for await (const chunk of session.streamReplyToLast({ mirror: false })) {
        rawReply += chunk;
      }
    } catch (e) {
      // Remove placeholder
      useXYData.setState({
        messages: useXYData.getState().messages.filter((m) => m.id !== placeholderId),
      });

      if (e instanceof AIUnavailableError) {
        // Mock fallback — unchanged behaviour
        return mockReplyFallback(convId, get);
      }

      const brief = e instanceof Error
        ? (e.message.length > 80 ? e.message.slice(0, 80) + '…' : e.message)
        : String(e);
      _appendMessage({
        id: uid(), convId,
        senderId: `char-${characterId}`,
        type: 'text', text: `[AI 回复失败] ${brief}`,
        timestamp: Date.now(),
      }, 'xingyu');
      return;
    } finally {
      if (aiSessions.get(convId) === session) aiSessions.delete(convId);
    }

    // Remove typing placeholder
    useXYData.setState({
      messages: useXYData.getState().messages.filter((m) => m.id !== placeholderId),
    });

    // parseReply → bubbles → deliver with delays (unchanged logic)
    const items = parseReply(rawReply);
    const active = isChatActive(convId);

    for (let i = 0; i < items.length; i++) {
      if (i > 0) {
        await new Promise((r) => setTimeout(r, 300 + Math.random() * 500));
      }
      const item = items[i]!;
      const ts = Date.now();

      if (item.type === 'signature') {
        useXYData.getState().updateCharacterSignature(characterId, item.text);
        continue;
      }

      let msg: Message;
      if (item.type === 'sticker') {
        const stickerPacks = useStickerStore.getState().packs;
        let foundSticker: { imageData: string; description: string } | undefined;
        for (const pack of stickerPacks) {
          const s = pack.stickers.find((st) => st.id === item.stickerId);
          if (s) { foundSticker = s; break; }
        }
        if (foundSticker) {
          msg = {
            id: uid(), convId,
            senderId: `char-${characterId}`,
            type: 'sticker',
            stickerUrl: foundSticker.imageData,
            stickerDesc: foundSticker.description,
            timestamp: ts,
          };
        } else {
          msg = {
            id: uid(), convId,
            senderId: `char-${characterId}`,
            type: 'text', text: item.content || '[表情]',
            timestamp: ts,
          };
        }
      } else {
        const text = filterReply(item.content) || '[空回复]';
        msg = {
          id: uid(), convId,
          senderId: `char-${characterId}`,
          type: 'text', text,
          timestamp: ts,
        };
      }

      // S3's _appendMessage would write a per-bubble entry to memoryStore —
      // we want ONE raw entry for the whole reply, not per bubble. Use
      // xingYuDataStore directly for the bubbles.
      const currentState = useXYData.getState();
      useXYData.setState({
        messages: [...currentState.messages, msg],
        conversations: currentState.conversations.map((c) =>
          c.id === convId
            ? { ...c, lastMsg: derivePreviewForBubble(msg), lastTime: ts,
                unread: active ? c.unread : c.unread + 1 }
            : c,
        ),
      });
    }

    // Write ONE raw assistant entry to memoryStore for the entire reply.
    useCharacterMemory.getState().append(characterId, {
      role: 'assistant',
      speakerId: characterId,
      content: rawReply,
      source: 'xingyu',
    });
  })();
}

function derivePreviewForBubble(msg: Message): string {
  switch (msg.type) {
    case 'text': return (msg.text ?? '').slice(0, 60);
    case 'sticker': return msg.stickerDesc ? `[表情：${msg.stickerDesc}]` : '[表情]';
    default: return '';
  }
}

function mockReplyFallback(convId: string, _get: () => XingYuDataState) {

  // logic from the old scheduleAICharacterReply and put it here unchanged.
  // Kept as a single function so it's easy to remove later when we add
  // real "AI unavailable" UX.
  // ...
}

// src/apps/XingYu/__tests__/xingyu-via-sdk.test.ts
import { describe, it, expect, vi, beforeEach } from 'vitest';
import { useXYData } from '../xingYuDataStore';
import { useCharacterStore } from '@/platform/stores/characterStore';
import { useCharacterMemory } from '@/platform/ai/characterMemoryStore';
import { useAIConfigStore } from '@/platform/stores/aiConfigStore';
import * as chatCompleteMod from '@/platform/ai/chatComplete';

beforeEach(() => {
  useCharacterMemory.getState().clearAll();
  useCharacterStore.setState({
    characters: [{ id: 'char-001', name: '小星', description: '', personality: '', scenario: '', systemPrompt: '', postHistoryInstructions: '', messageExamples: '' } as never],
  });
  useXYData.setState({
    conversations: [{ id: 'c-char-001', idolId: 'char-001', characterId: 'char-001', lastMsg: '', lastTime: 0, unread: 0 }],
    messages: [], moments: [], interactions: [],
  });
  useAIConfigStore.setState({
    apiKey: 'sk', model: 'x', provider: 'openai', apiEndpoint: 'e',
    contextWindow: 100_000, maxTokens: 2000, summarizeThreshold: 0,
  } as never);
});

describe('XingYu text chat via SDK session', () => {
  it('sends user msg → AI reply JSON gets parsed into bubbles AND raw goes to memoryStore', async () => {
    const rawJson = '[{"type":"text","content":"挺不错的呀"},{"type":"text","content":"阳光明媚"}]';
    vi.spyOn(chatCompleteMod, 'chatComplete').mockResolvedValue(rawJson);

    useXYData.getState().sendMessage('c-char-001', '今天天气怎么样');

    // Wait for bubble delivery (at most 2 bubbles × ~800ms + buffer)
    await new Promise((r) => setTimeout(r, 2500));

    // xingYuDataStore: the user msg + 2 bubble replies (no placeholder remaining)
    const msgs = useXYData.getState().messages.filter((m) => !m.streaming);
    expect(msgs.map((m) => m.type === 'text' ? m.text : m.type)).toEqual([
      '今天天气怎么样', '挺不错的呀', '阳光明媚',
    ]);

    // memoryStore: user msg entry + single raw-JSON assistant entry
    const mem = useCharacterMemory.getState().getAll('char-001');
    expect(mem).toHaveLength(2);
    expect(mem[0]!).toMatchObject({ role: 'user', speakerId: 'me', content: '今天天气怎么样' });
    expect(mem[1]!).toMatchObject({ role: 'assistant', speakerId: 'char-001', content: rawJson });
  });
});
