// src/platform/ai/__tests__/m4.1.e2e.test.ts
import { describe, it, expect, vi, beforeEach } from 'vitest';
import {
  chatWithCharacter,
} from '@/platform/userApp/sdk/ai';
import { useCharacterMemory } from '../characterMemoryStore';
import { useCharacterStore } from '@/platform/stores/characterStore';
import { useAIConfigStore } from '@/platform/stores/aiConfigStore';
import { usePersonaStore } from '@/platform/stores/personaStore';
import * as chatCompleteMod from '../chatComplete';
import { withUserAppContext } from '@/platform/userApp/sdk/context';

beforeEach(() => {
  useCharacterMemory.getState().clearAll();
  useCharacterStore.setState({
    characters: [{ id: 'char-001', name: '小星', description: '', personality: '', scenario: '', systemPrompt: '', postHistoryInstructions: '', messageExamples: '' } as never],
  });
  usePersonaStore.setState({ /* getActivePersona returns { name: '小米', description: '' } */ } as never);
  useAIConfigStore.setState({
    apiKey: 'sk', model: 'x', provider: 'openai', apiEndpoint: 'e',
    contextWindow: 100_000, maxTokens: 2000, summarizeThreshold: 0,
  } as never);
});

describe('M4.1 E2E — persistent vs clone semantics across sessions', () => {
  it('persistent=true: two sessions share memory', async () => {
    vi.spyOn(chatCompleteMod, 'chatComplete')
      .mockResolvedValueOnce('first reply')
      .mockResolvedValueOnce('second reply');

    await withUserAppContext('app-test', async () => {
      const s1 = chatWithCharacter('char-001', { persistent: true });
      await s1.send('hello');
    });

    // Simulate "app closed + reopened" — new session with same character
    await withUserAppContext('app-test', async () => {
      const s2 = chatWithCharacter('char-001', { persistent: true });
      expect(s2.history).toEqual([]); // session buffer is fresh

      // But memoryStore retains the previous conversation
      const mem = useCharacterMemory.getState().getAll('char-001');
      expect(mem).toHaveLength(2); // user + assistant from s1
      expect(mem[0]!.content).toBe('hello');
      expect(mem[1]!.content).toBe('first reply');

      // The next send's prompt includes the earlier exchange
      await s2.send('are you still there');
      const finalMem = useCharacterMemory.getState().getAll('char-001');
      expect(finalMem).toHaveLength(4);
    });
  });

  it('persistent=false: clone memory is not retained across sessions', async () => {
    vi.spyOn(chatCompleteMod, 'chatComplete').mockResolvedValue('clone reply');

    await withUserAppContext('app-test', async () => {
      const s1 = chatWithCharacter('char-001', { persistent: false });
      await s1.send('secret');
    });

    expect(useCharacterMemory.getState().getAll('char-001')).toHaveLength(0);

    await withUserAppContext('app-test', async () => {
      const s2 = chatWithCharacter('char-001', { persistent: false });
      expect(s2.history).toEqual([]);
    });
  });
});
```

        systemPrompt: '', postHistoryInstructions: '',
        contextWindow: 100_000, maxTokens: 2000,

      },
      memoryEntries: useCharacterMemory.getState().getAll('char-001'),
      currentCharId: 'char-001',

// src/platform/ai/__tests__/aichat-cross-write.test.ts
import { describe, it, expect, beforeEach } from 'vitest';
import { _appendMessage } from '../memoryWriter';
import { useCharacterMemory } from '../characterMemoryStore';
import { useXYData } from '@/apps/XingYu/xingYuDataStore';
import { useCharacterStore } from '@/platform/stores/characterStore';
import { usePersonaStore } from '@/platform/stores/personaStore';
import type { Message } from '@/apps/XingYu/data';

beforeEach(() => {
  useCharacterMemory.getState().clearAll();
  useCharacterStore.setState({
    characters: [
      { id: 'char-a', name: '小星' } as never,
      { id: 'char-b', name: '小月' } as never,
    ],
  });
  usePersonaStore.setState({} as never);
  useXYData.setState({
    conversations: [{
      id: 'aiai-1',
      idolId: 'char-a',
      characterId: 'char-a',
      aiChatParticipants: ['char-a', 'char-b'],
      lastMsg: '', lastTime: 0, unread: 0,
    }],
    messages: [], moments: [], interactions: [],
  });
});

describe('AI-AI cross-write', () => {
  it('A speaks → A stores assistant, B stores user, content identical', () => {
    const msgA: Message = {
      id: 'm1', convId: 'aiai-1', senderId: 'char-a',
      type: 'text', text: '你好啊', timestamp: 1,
    };
    _appendMessage(msgA, 'xingyu');

    const aMem = useCharacterMemory.getState().getAll('char-a');
    const bMem = useCharacterMemory.getState().getAll('char-b');
    expect(aMem).toHaveLength(1);
    expect(bMem).toHaveLength(1);

    expect(aMem[0]!).toMatchObject({
      role: 'assistant', speakerId: 'char-a', content: '你好啊',
    });
    expect(bMem[0]!).toMatchObject({
      role: 'user', speakerId: 'char-a', content: '你好啊',
    });
  });

  it('B replies → mirror flipped from each POV', () => {
    _appendMessage({
      id: 'm1', convId: 'aiai-1', senderId: 'char-a',
      type: 'text', text: 'A-line', timestamp: 1,
    }, 'xingyu');
    _appendMessage({
      id: 'm2', convId: 'aiai-1', senderId: 'char-b',
      type: 'text', text: 'B-line', timestamp: 2,
    }, 'xingyu');

    const aMem = useCharacterMemory.getState().getAll('char-a');
    expect(aMem.map((e) => ({ role: e.role, content: e.content }))).toEqual([
      { role: 'assistant', content: 'A-line' },
      { role: 'user',      content: 'B-line' },
    ]);

    const bMem = useCharacterMemory.getState().getAll('char-b');
    expect(bMem.map((e) => ({ role: e.role, content: e.content }))).toEqual([
      { role: 'user',      content: 'A-line' },
      { role: 'assistant', content: 'B-line' },
    ]);
  });
});
```

# M4.1 Release Notes — AI SDK + XingYu migration

**Date:** 2026-04-19
**Status:** accepted

## What landed

- `@hiphone/ai` SDK (Level 1 + Level 2): `complete` / `streamComplete` /
  `chatWithCharacter` (session-based) / `getCharacters` / `extractPlainText`
- New platform layer `characterMemoryStore` — per-character memory stream,
  raw content, speakerId, IDB persistence
- `promptAssembly` switched from first-person labeled system block to
  role alternation with "<speaker>：" prefix on user content; raw JSON
  assistant replies preserved byte-identically
- XingYu fully migrated: `sendMessage` / `sendImage` / `sendSticker` /
  `sendForwardCard` / `sendNote` / `sendSong` all route through the new
  session + memoryStore. AbortController map → sessions map. Bubble
  delivery, filterReply, IDOL_REPLY_POOL mock fallback all preserved.
- Heartbeat: data source switched (reads memoryStore; writes via
  `_appendMessage`). ReAct protocol + 15 tools unchanged.
- aiChatEngine: writes via `_appendMessage` — AI-AI conversations
  auto-fan-out to both participants' memoryStores.
- Two demo fixtures: `ai-translator-app` (Level 1) + `ai-chat-app` (Level 2).

## UX changes users should know

### ⚠️ AI characters "forget once" on first boot after upgrade

M4.1 moves the AI memory layer from inline xingYuDataStore to a new
`characterMemoryStore` and does NOT migrate existing messages. The first
time you open the app after this release, each character starts with an
empty memory. All prior XingYu conversations remain in the UI but are
not visible to the AI in new prompts.

This is intentional — migration tooling was deemed higher risk than UX
cost. The rebuild starts naturally: each new message from this point
forward populates the memory, and summarisation kicks in once the
threshold is crossed, same as before.

### Prompt format change

Internally, prompt history is now OpenAI-native role alternation (user /
assistant turns) instead of the one-big-system-block "labeled narration"
we used before. Each user-role content is prefixed with the speaker's
display name (e.g. `小米：你好`). assistant-role content is passed to
the LLM verbatim, so the model now sees its own prior JSON emissions
byte-for-byte (previously these were flattened into separate "assistant"
turns per bubble, losing `stickerId` and `signature` action info).

Side effect: models are more likely to keep stickerId consistent across
turns and to recall signature updates they performed, because the raw
traces are now in the prompt.
